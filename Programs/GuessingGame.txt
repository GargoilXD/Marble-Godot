Function List Make_random_sample(Integer Size, Integer lower, Integer upper) {
	List sample = Range(Size)
	For Integer x in Range(Size) {
		sample[x] = Random(lower, upper)
	}
	Return sample
}
Function String Output_list(List Data) {
	String Output = ''
	For Variant x in Data {
		Output += x + ', '
	}
	Return Output
}

List Computer_numbers = Make_random_sample(5, 0, 10)
Integer Size = 5
List Responses = Range(Size)

For Integer x in Range(Size) {
	Responses[x] = Input('Guess a number:')
}

Integer number_matched = 0
For Integer x in Range(Size) {
	For Integer y in Range(Size) {
		if Responses[x] == (Computer_numbers[y] + '') {
			number_matched += 1
		}
	}
}

Print('The computer chose:', Output_list(Computer_numbers))
Print('You chose:', Output_list(Responses))
Print('There were', number_matched, 'Matches')
