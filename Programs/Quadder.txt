Integer A = 1
Integer B = 6
Integer C = 8
Float D = B * B - 4 * A * C

Variant X1 = 0
Variant X2 = 0
if(D > 0){
	X1 = (-B - D ^ 0.5)/(2 * A)
	X2 = (-B + D ^ 0.5)/(2 * A)
}
elseif(D == 0){
	X1 = -B/(2 * A)
	X2 = X2
}
else{
	X1 = -B/(2 * A) + " - " + (-D ^ 0.5)/(2 * A) + "i"
	X2 = -B/(2 * A) + " + " + (-D ^ 0.5)/(2 * A) + "i"
}
Print("X1: ", X1, "X2: ", X2)
